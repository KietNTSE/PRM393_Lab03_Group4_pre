void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // 1. Khởi tạo Firebase từ DefaultFirebaseOptions
  await Firebase.initializeApp(
    options: DefaultFirebaseOptions.currentPlatform,
  );

  // 2. Thiết lập Crashlytics bắt lỗi toàn cục
  FlutterError.onError = FirebaseCrashlytics.instance.recordFlutterFatalError;
  PlatformDispatcher.instance.onError = (error, stack) {
    FirebaseCrashlytics.instance.recordError(error, stack, fatal: true);
    return true;
  };

  // 3. Khởi tạo FCM Background Handler & Permission
  FirebaseMessaging.onBackgroundMessage(_firebaseMessagingBackgroundHandler);

  // 4. Remote Config Fetch ban đầu (1 giờ interval cho prod)
  await RemoteConfigService().initialize();

  runApp(const ProviderScope(child: EventHubApp()));
}

Tích Hợp AuthService & State Management

Code AuthService Thực Tế
class AuthService {
  final FirebaseAuth _auth = FirebaseAuth.instance;

  // Reactive stream cho Riverpod Provider
  Stream<User?> get authStateChanges => _auth.authStateChanges();

  User? get currentUser => _auth.currentUser;

  Future<UserCredential> signIn(String email, String password) async {
    try {
      final cred = await _auth.signInWithEmailAndPassword(
        email: email.trim(),
        password: password,
      );
      // Gán user_id dạng hash cho Analytics & Crashlytics
      await AnalyticsService().setUserId(cred.user?.uid);
      await CrashlyticsService().setUserId(cred.user?.uid);
      return cred;
    } on FirebaseAuthException catch (e) {
      throw AuthFailure(e.code, e.message);
    }
  }
}
Quản Lý Luồng Đăng Nhập
  • authStateProvider: StreamProvider lắng nghe Firebase Auth thay đổi để rebuild App Router tức thì.
  • Token Interceptor: Gắn Authorization: Bearer <idToken> vào header các request REST API.
  • Sign Out Sanitization: Xóa sạch cache, session tokens và unregister FCM token khi đăng xuất.

Quy Chuẩn Security Rules Phân Quyền UID

Bảo vệ dữ liệu người dùng tuyệt đối trong tệp storage.rules: 

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Chỉ người dùng đã đăng nhập mới được thao tác
    // Chỉ đọc/ghi đúng thư mục UID của chính mình
    match /users/{userId}/exports/{fileName} {
      allow read, write: if request.auth != null 
                         && request.auth.uid == userId
                         && request.resource.size < 5 * 1024 * 1024; // Max 5MB
    }
  }
}

Kiến Trúc & 3 Trạng Thái Hoạt Động FCM

1. Foreground State

App đang mở và tương tác trên màn hình.

  • Tự tạo Local Notification Heads-up Banner

2. Background State

App bị thu nhỏ (người dùng bấm Home / qua app khác).

  • Xử lý ngầm qua onBackgroundMessage
  • Android OS hiển thị thông báo trên Notification Tray
  • Bấm vào thông báo → Mở deep link vào Inbox Tab
3. Terminated State

App đã bị đóng hoàn toàn (Force Stop / Vuốt tắt).

  • OS nhận thông báo từ Google Play Services
  • Bắt Intent qua getInitialMessage()
  • Điều hướng chính xác tới chi tiết vé hoặc sự kiện

@pragma('vm:entry-point')
Future<void> _firebaseMessagingBackgroundHandler(RemoteMessage message) async {
  await Firebase.initializeApp();
  // Xử lý payload và lưu vào SQLite Local Storage
  final db = await DatabaseHelper.instance.database;
  await db.insert('notifications', {
    'id': message.messageId,
    'title': message.notification?.title ?? 'EventHub',
    'body': message.notification?.body ?? '',
    'created_at': DateTime.now().toIso8601String(),
    'read': 0,
  });
}
Xin Quyền Android 13+ & Routing
Future<void> setupFCM(BuildContext context) async {
  // Xin quyền người dùng Android 13+
  NotificationSettings settings = await FirebaseMessaging.instance.requestPermission(
    alert: true, badge: true, sound: true,
  );

  // Xử lý trường hợp App mở từ Terminated State
  RemoteMessage? initialMessage = 
      await FirebaseMessaging.instance.getInitialMessage();
  if (initialMessage != null) {
    _handleMessageNavigation(initialMessage);
  }

  // Lắng nghe khi bấm notification từ Background
  FirebaseMessaging.onMessageOpenedApp.listen(_handleMessageNavigation);
}

  • user_id: Mã UID ẩn danh của người dùng
  • custom_logs: Chuỗi hành động gần nhất của user

CrashlyticsService Toàn Cục
class CrashlyticsService {
  final FirebaseCrashlytics _crashlytics = FirebaseCrashlytics.instance;

  // Ghi nhận lỗi Non-fatal kèm Context Keys
  Future<void> recordNonFatal({
    required dynamic exception,
    required StackTrace stack,
    required String reason,
    Map<String, Object>? customKeys,
  }) async {
    if (customKeys != null) {
      for (final entry in customKeys.entries) {
        await _crashlytics.setCustomKey(entry.key, entry.value);
      }
    }
    await _crashlytics.recordError(
      exception,
      stack,
      reason: reason,
      fatal: false,
    );
  }
}

Tích Hợp RemoteConfigService & Cache Policy

Code Tích Hợp Đầy Đủ
class RemoteConfigService {
  final FirebaseRemoteConfig _rc = FirebaseRemoteConfig.instance;

  Future<void> initialize() async {
    // 1. Đặt giá trị mặc định tránh null khi offline
    await _rc.setDefaults({
      'recommendation_limit': 10,
      'show_recommendations': true,
    });

    // 2. Cấu hình Cache Expiration linh hoạt
    await _rc.setConfigSettings(RemoteConfigSettings(
      fetchTimeout: const Duration(seconds: 10),
      minimumFetchInterval: kDebugMode 
          ? Duration.zero              // Realtime trong lúc dev/test
          : const Duration(hours: 1),  // 1 giờ cho production
    ));

    // 3. Fetch và kích hoạt cấu hình mới nhất
    await _rc.fetchAndActivate();
  }
}
Tiêu Thụ Trong Presentation Widget
// ConsumerWidget đọc qua Riverpod
Widget build(BuildContext context, WidgetRef ref) {
  final showRecs = ref.watch(showRecommendationsProvider);
  final limit = ref.watch(recommendationLimitProvider);

  if (!showRecs) {
    return const SizedBox.shrink(); // Ẩn hoàn toàn layout
  }

  return RecommendationCarousel(
    limit: limit,
    events: events.take(limit).toList(),
  );
}
